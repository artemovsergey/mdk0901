# Самостоятельная работа 5. Работа с Yandex Maps

## Тема
Создание интерактивных карт с использованием Yandex Maps API

## Цель
Освоить работу с Yandex Maps JavaScript API: научиться создавать карты, добавлять метки, кластеризацию, геокодирование, маршруты и кастомные элементы.

## Теоретическая часть

### Основные концепции Yandex Maps API:

1. **Карта (Map)** - основной объект для отображения карты
2. **Метки (Placemarks)** - точки на карте
3. **Кластеризация** - группировка близко расположенных меток
4. **Геокодирование** - преобразование адреса в координаты и обратно
5. **Маршрутизация** - построение маршрутов между точками
6. **Кастомные балуны** - всплывающие окна с информацией

### Ключевые классы и методы:

```javascript
// Создание карты
const map = new ymaps.Map('map', {
    center: [55.76, 37.64],
    zoom: 10
});

// Создание метки
const placemark = new ymaps.Placemark([55.76, 37.64], {
    balloonContent: 'Описание метки'
});

// Добавление метки на карту
map.geoObjects.add(placemark);
```

### Типы карт:
- `yandex#map` - схема
- `yandex#satellite` - спутник
- `yandex#hybrid` - гибрид

## Практический пример

```javascript
// Инициализация карты
ymaps.ready(init);

function init() {
    const map = new ymaps.Map('map', {
        center: [55.76, 37.64],
        zoom: 10
    });

    // Добавление метки
    const placemark = new ymaps.Placemark([55.76, 37.64], {
        balloonContent: '<strong>Москва</strong><br>Столица России'
    }, {
        preset: 'islands#icon',
        iconColor: '#0095b6'
    });

    map.geoObjects.add(placemark);
}
```

class LandmarksMap {
    constructor() {
        this.map = null;
        this.placemarks = [];
        this.objectManager = null;
        this.filteredPlaces = [...landmarksData];
        this.init();
    }

    init() {
        ymaps.ready(() => {
            this.createMap();
            this.addPlacemarks();
            this.setupEventListeners();
            this.updateStatistics();
            this.renderPlacesList();
        });
    }

    createMap() {
        this.map = new ymaps.Map('map', {
            center: [55.7558, 37.6173],
            zoom: 10,
            controls: ['zoomControl', 'fullscreenControl']
        });

        // Добавляем поиск
        const searchControl = new ymaps.control.SearchControl({
            options: {
                provider: 'yandex#search',
                noPlacemark: true
            }
        });
        this.map.controls.add(searchControl);

        // Инициализация ObjectManager для кластеризации
        this.objectManager = new ymaps.ObjectManager({
            clusterize: true,
            gridSize: 64,
            clusterDisableClickZoom: true
        });

        this.objectManager.objects.options.set('preset', 'islands#blueDotIcon');
        this.objectManager.clusters.options.set('preset', 'islands#blueClusterIcons');
        this.map.geoObjects.add(this.objectManager);
    }

    addPlacemarks() {
        const objects = landmarksData.map(place => ({
            id: place.id,
            type: 'Feature',
            geometry: {
                type: 'Point',
                coordinates: place.coordinates
            },
            properties: {
                balloonContent: this.createBalloonContent(place),
                clusterCaption: place.name,
                hintContent: place.name
            },
            options: {
                preset: this.getPresetByCategory(place.category)
            }
        }));

        this.objectManager.add(objects);
    }

    createBalloonContent(place) {
        return `
            <div class="balloon">
                <div class="balloon-title">${place.name}</div>
                <div class="balloon-rating">★ ${place.rating}</div>
                ${place.image ? `<img src="${place.image}" alt="${place.name}" class="balloon-image" onerror="this.style.display='none'">` : ''}
                <div class="balloon-description">${place.description}</div>
                <div class="balloon-address">📍 ${place.address}</div>
                ${place.website ? `<a href="${place.website}" target="_blank" class="balloon-link">Подробнее</a>` : ''}
            </div>
        `;
    }

    getPresetByCategory(category) {
        const presets = {
            historical: 'islands#redIcon',
            cultural: 'islands#violetIcon',
            religious: 'islands#orangeIcon',
            parks: 'islands#greenIcon',
            museums: 'islands#blueIcon'
        };
        return presets[category] || 'islands#blueIcon';
    }

    setupEventListeners() {

        document.getElementById('categoryFilter').addEventListener('change', (e) => {
            this.applyFilters();

        document.getElementById('ratingFilter').addEventListener('change', (e) => {

            document.getElementById('categoryFilter').value = 'all';
            document.getElementById('ratingFilter').value = '0';

            this.getCurrentLocation();
        });

        // Кнопка общего вида
        document.getElementById('resetView').addEventListener('click', () => {
            this.map.setBounds(this.map.geoObjects.getBounds());
        });

        // Обработка кликов на объекты ObjectManager
        this.objectManager.objects.events.add('click', (e) => {
            const objectId = e.get('objectId');
            const place = landmarksData.find(p => p.id == objectId);
            if (place) {
                this.highlightPlaceInList(place.id);
            }
        });
    }

    applyFilters() {
        const categoryFilter = document.getElementById('categoryFilter').value;
        const ratingFilter = parseFloat(document.getElementById('ratingFilter').value);

        this.filteredPlaces = landmarksData.filter(place => {
            const categoryMatch = categoryFilter === 'all' || place.category === categoryFilter;
            const ratingMatch = ratingFilter === 0 || place.rating >= ratingFilter;
            return categoryMatch && ratingMatch;
        });

        this.updateMap();
        this.renderPlacesList();
        this.updateStatistics();
    }

    searchPlaces() {
        const searchTerm = document.getElementById('searchInput').value.toLowerCase().trim();
        
        if (!searchTerm) {
            this.applyFilters();
            return;
        }

        this.filteredPlaces = landmarksData.filter(place =>
            place.name.toLowerCase().includes(searchTerm) ||
            place.description.toLowerCase().includes(searchTerm) ||
            place.address.toLowerCase().includes(searchTerm)
        );

        this.updateMap();
        this.renderPlacesList();
        this.updateStatistics();

        // Если найден один результат, центрируем карту на нём
        if (this.filteredPlaces.length === 1) {
            this.map.setCenter(this.filteredPlaces[0].coordinates, 15);
        }
    }

    updateMap() {
        // Очищаем ObjectManager
        this.objectManager.removeAll();
        
        // Добавляем отфильтрованные объекты
        const objects = this.filteredPlaces.map(place => ({
            id: place.id,
            type: 'Feature',
            geometry: {
                type: 'Point',
                coordinates: place.coordinates
            },
            properties: {
                balloonContent: this.createBalloonContent(place),
                clusterCaption: place.name,
                hintContent: place.name
            },
            options: {
                preset: this.getPresetByCategory(place.category)
            }
        }));

        this.objectManager.add(objects);

        // Автоматически подбираем масштаб, если есть объекты
        if (this.filteredPlaces.length > 0) {
            const bounds = this.objectManager.getBounds();
            if (bounds) {
                this.map.setBounds(bounds, { checkZoomRange: true });
            }
        }
    }

    renderPlacesList() {
        const container = document.getElementById('placesContainer');
        
        if (this.filteredPlaces.length === 0) {
            container.innerHTML = '<p style="text-align: center; color: #7f8c8d;">Места не найдены</p>';
            return;
        }

        container.innerHTML = this.filteredPlaces.map(place => `
            <div class="place-item ${place.category}" onclick="landmarksMap.flyToPlace(${place.id})">
                <div class="place-header">
                    <div class="place-name">${place.name}</div>
                    <div class="place-rating">★ ${place.rating}</div>
                </div>
                <div class="place-category">${this.getCategoryName(place.category)}</div>
                <div class="place-address">📍 ${place.address}</div>
                <div class="place-description">${place.description}</div>
            </div>
        `).join('');
    }

    getCategoryName(category) {
        const categories = {
            historical: 'Историческое место',
            cultural: 'Культурный объект',
            religious: 'Религиозный объект',
            parks: 'Парк',
            museums: 'Музей'
        };
        return categories[category] || category;
    }

    flyToPlace(placeId) {
        const place = landmarksData.find(p => p.id === placeId);
        if (place) {
            this.map.setCenter(place.coordinates, 16, {
                duration: 1000
            });
            
            // Открываем балун
            this.objectManager.objects.balloon.open(placeId);
            
            this.highlightPlaceInList(placeId);
        }
    }

    highlightPlaceInList(placeId) {
        // Убираем выделение со всех элементов
        document.querySelectorAll('.place-item').forEach(item => {
            item.style.background = 'white';
        });
        
        // Выделяем выбранный элемент
        const selectedItem = document.querySelector(`.place-item[onclick*="${placeId}"]`);
        if (selectedItem) {
            selectedItem.style.background = '#e3f2fd';
        }
    }

    getCurrentLocation() {
        if (!navigator.geolocation) {
            alert('Геолокация не поддерживается вашим браузером');
            return;
        }

        navigator.geolocation.getCurrentPosition(
            (position) => {
                const userLocation = [position.coords.latitude, position.coords.longitude];
                
                // Добавляем метку пользователя
                const userPlacemark = new ymaps.Placemark(userLocation, {
                    balloonContent: 'Ваше местоположение'
                }, {
                    preset: 'islands#greenDotIcon'
                });

                this.map.geoObjects.add(userPlacemark);
                this.map.setCenter(userLocation, 15);

                // Находим ближайшую достопримечательность
                this.findNearestLandmark(userLocation);
            },
            (error) => {
                alert('Не удалось определить ваше местоположение');
            }
        );
    }

    findNearestLandmark(userLocation) {
        let nearestPlace = null;
        let minDistance = Infinity;

        landmarksData.forEach(place => {
            const distance = this.calculateDistance(userLocation, place.coordinates);
            if (distance < minDistance) {
                minDistance = distance;
                nearestPlace = place;
            }
        });

        if (nearestPlace) {
            setTimeout(() => {
                alert(`Ближайшая достопримечательность: ${nearestPlace.name} (${Math.round(minDistance)} км)`);
            }, 1000);
        }
    }

    calculateDistance(coord1, coord2) {
        const [lat1, lon1] = coord1;
        const [lat2, lon2] = coord2;
        
        const R = 6371; // Радиус Земли в км
        const dLat = (lat2 - lat1) * Math.PI / 180;
        const dLon = (lon2 - lon1) * Math.PI / 180;
        
        const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
                Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
                Math.sin(dLon/2) * Math.sin(dLon/2);
        
        const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
        return R * c;
    }

    updateStatistics() {
        document.getElementById('totalPlaces').textContent = landmarksData.length;
        document.getElementById('visiblePlaces').textContent = this.filteredPlaces.length;
        
        const averageRating = this.filteredPlaces.length > 0 
            ? (this.filteredPlaces.reduce((sum, place) => sum + place.rating, 0) / this.filteredPlaces.length).toFixed(1)
            : '0.0';
        document.getElementById('averageRating').textContent = averageRating;
    }
}

    landmarksMap = new LandmarksMap();
